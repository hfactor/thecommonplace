{{- /* dict: content (string), canon (dict title->{url,preview}), notes (page slice), deadPlain (bool, optional) — true renders unresolved [[..]] as plain text instead of a styled "dead" link (used for footnote citations to books/sources that aren't notes in this site). */ -}} {{- $content := .content -}} {{- $canon := .canon -}} {{- $notes := .notes -}} {{- $deadPlain := .deadPlain | default false -}} {{- range $notes -}} {{- $target := . -}} {{- $info := index $canon $target.Title -}} {{- if $info -}} {{- $pattern := printf `\[\[%s\]\]` (replaceRE `([.+*?()\[\]{}|^$\\])` `\$1` $target.Title) -}} {{- $replacement := printf `%s` $info.url (htmlEscape $info.preview) $target.Title -}} {{- $content = replaceRE $pattern $replacement $content -}} {{- end -}} {{- end -}} {{- if $deadPlain -}} {{- $content = replaceRE `\[\[([^\]]+)\]\]` `$1` $content -}} {{- else -}} {{- $content = replaceRE `\[\[([^\]]+)\]\]` `$1` $content -}} {{- end -}} {{- $content | safeHTML -}}
