{{- /* dict: content (string), canon (dict title->{url,preview}), notes (page slice) */ -}} {{- $content := .content -}} {{- $canon := .canon -}} {{- $notes := .notes -}} {{- range $notes -}} {{- $target := . -}} {{- $info := index $canon $target.Title -}} {{- if $info -}} {{- $pattern := printf `\[\[%s\]\]` (replaceRE `([.+*?()\[\]{}|^$\\])` `\$1` $target.Title) -}} {{- $replacement := printf `%s` $info.url (htmlEscape $info.preview) $target.Title -}} {{- $content = replaceRE $pattern $replacement $content -}} {{- end -}} {{- end -}} {{- $content = replaceRE `\[\[([^\]]+)\]\]` `$1` $content -}} {{- $content | safeHTML -}}
